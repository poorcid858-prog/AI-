### Step 1: 后端 — 模型配置CRUD

在 `lib/admin-config.js` 追加：

```js
// ========== 模型配置 ==========
function listModels() {
  return store.read('models', []);
}
function getModel(id) {
  return listModels().find(m => m.id === id);
}
function createModel(data) {
  const id = store.nextId('models', 'm');
  const model = { id, ...data, enabled: true, createdAt: new Date().toISOString() };
  store.push('models', model);
  return model;
}
function updateModel(id, data) {
  return store.update('models', id, data);
}
function deleteModel(id) {
  return store.remove('models', id);
}
```

模型字段：`id, name, type, apiUrl, apiKey, temperature, maxTokens, enabled, createdAt`
