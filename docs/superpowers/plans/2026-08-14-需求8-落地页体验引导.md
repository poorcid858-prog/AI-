- [ ] **Step 2: 添加 `quickStart` 函数到 index.html 的 `<script>` 中**

在 `$('#loginForm').addEventListener('submit', ...)` 之前添加：

```javascript
// ---- 三步引导快捷入口 ----
function quickStart(role, question) {
  // 找到对应角色的演示账号
  const accounts = document.querySelectorAll('.acct');
  let target = null;
  for (const el of accounts) {
    if (el.dataset.username === role || el.dataset.username.includes(role)) {
      target = el;
      break;
    }
  }
  if (target) {
    target.click(); // 自动选中账号
    // 如果第 3 步（审计页），直接登录后跳转
    if (role === 'admin' && !question) {
      // 选中管理员账号后自动登录跳转到审计页
      doLogin('/admin-qa.html');
    } else {
      doLogin('/workspace.html?q=' + encodeURIComponent(question));
    }
  }
}

async function doLogin(redirect) {
  const btn = $('#submitBtn');
  btn.disabled = true;
  btn.textContent = '登录中…';
  try {
    const res = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ username: $('#username').value.trim(), password: $('#password').value }),
    });
    const d = await res.json();
    if (!d.ok) throw new Error(d.error || '登录失败');
    localStorage.setItem('token', d.token);
    localStorage.setItem('user', JSON.stringify(d.user));
    localStorage.setItem('permissions', JSON.stringify(d.permissions));
    // 存储用户角色到 sessionStorage，供工作台使用
    sessionStorage.setItem('userRole', d.user.role);
    location.href = redirect;
  } catch (e) {
    showErr(e.message);
  } finally {
    btn.disabled = false;
    btn.textContent = '登 录';
  }
}
```
