# 需求 5：AI 能力中心可编辑 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 实现 AI 能力中心中 Skill 和 Prompt 的编辑、草稿保存、试跑、新旧版本对比、发布生效、弃稿、版本管理与回滚、全程审计留痕。

**Architecture:** 复用 `lib/knowledge-layers.js` 的版本管理状态机模式（草稿→发布→回滚），但独立于四层文档模型，因为能力编辑是独立领域。能力数据（Skill/Prompt 定义）存为 JSON 文件，每个能力有草稿版和生效版，草稿版可编辑和试跑，满意后发布替换生效版。试跑走已有 `lib/prompt-engine.js` 的 `assemblePrompt` 流程，用真实历史问题生成输出，与生效版本结果并排对比。

**Tech Stack:** Node.js + Express + JSON 文件存储（`lib/store.js`）+ 复用 `lib/prompt-engine.js` + `lib/rag-engine.js`

**Spec:** `docs/需求5-AI能力中心可编辑.md`

## 全局约束

- 使用 `lib/store.js` 做 JSON 文件存储（复用 `data/` 目录）
- 使用 `lib/prompt-engine.js` 的 `assemblePrompt` 和 `SKILLS` 做试跑
- 复用 `lib/rag-engine.js` 做试跑时的 RAG 检索
- 使用 `node --test` 测试框架，测试文件放 `test/` 目录下
- 所有新文件放在 `lib/`、`routes/`、`public/`、`test/` 对应目录
- 在 `server.js` 注册新的路由模块
- 所有修改操作记录审计日志（`data/capability-audit.json`）
- 保留最近 N 个生效版本，可回滚

```javascript
// lib/capability-engine.js
"use strict";
const store = require('./store');
const config = require('../config');
const promptEngine = require('./prompt-engine');

const CAPABILITIES_FILE = 'capabilities';
const AUDIT_FILE = 'capability-audit';
const DEFAULT_MAX_HISTORY = 10;

// 初始化默认能力（从 prompt-engine.js 读 skill 定义）
function getDefaultCapabilities() {
  const skills = promptEngine.SKILLS;
  return Object.entries(skills).map(([role, skill]) => ({
    id: `skill_${role}`,
    type: 'skill',
    name: `${skill.title} Skill`,
    description: `${skill.title}的角色设定与任务指令`,
    published: {
      version: 1,
      content: skill,
      publishedAt: new Date().toISOString(),
      publishedBy: 'system',
    },
    draft: null,
    history: [],
    maxHistory: DEFAULT_MAX_HISTORY,
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
  }));
}

// 读取所有能力
function listCapabilities() {
  const caps = store.read(CAPABILITIES_FILE, []);
  if (caps.length === 0) {
    const defaults = getDefaultCapabilities();
    store.write(CAPABILITIES_FILE, defaults);
    return defaults;
  }
  return caps;
}

// 读取单个能力
function getCapability(id) {
  const caps = listCapabilities();
  return caps.find(c => c.id === id) || null;
}

// 写回能力的辅助函数
function saveCapabilities(caps) {
  store.write(CAPABILITIES_FILE, caps);
}

// 获取能力列表（不含草稿细节，给前端用）
function listCapabilitySummaries() {
  return listCapabilities().map(c => ({
    id: c.id, type: c.type, name: c.name, description: c.description,
    hasDraft: c.draft !== null,
    publishedVersion: c.published ? c.published.version : null,
    draftVersion: c.draft ? c.draft.version : null,
    updatedAt: c.updatedAt,
  }));
}
```

```javascript
// 追加到 lib/capability-engine.js

// 记录审计日志
function appendAuditLog(entry) {
  const logs = store.read(AUDIT_FILE, []);
  logs.push({
    ...entry,
    timestamp: new Date().toISOString(),
  });
  store.write(AUDIT_FILE, logs);
}

// 编辑草稿：保存为新草稿或更新已有草稿
function editDraft(capId, content, editedBy) {
  const caps = listCapabilities();
  const idx = caps.findIndex(c => c.id === capId);
  if (idx === -1) throw Object.assign(new Error('能力不存在'), { status: 404 });

  const cap = caps[idx];
  const newVersion = cap.published ? cap.published.version + 1 : 1;
  cap.draft = {
    version: newVersion,
    content: JSON.parse(JSON.stringify(content)), // 深拷贝
    updatedAt: new Date().toISOString(),
  };
  cap.updatedAt = new Date().toISOString();
  caps[idx] = cap;
  saveCapabilities(caps);

  appendAuditLog({
    action: 'edit_draft',
    capId, editedBy,
    version: newVersion,
    detail: '保存草稿',
  });

  return cap;
}

// 获取草稿内容
function getDraft(capId) {
  const cap = getCapability(capId);
  if (!cap) throw Object.assign(new Error('能力不存在'), { status: 404 });
  return cap.draft;
}

// 获取生效版内容
function getPublished(capId) {
  const cap = getCapability(capId);
  if (!cap) throw Object.assign(new Error('能力不存在'), { status: 404 });
  return cap.published;
}

// 弃稿：删除草稿
function discardDraft(capId, editedBy) {
  const caps = listCapabilities();
  const idx = caps.findIndex(c => c.id === capId);
  if (idx === -1) throw Object.assign(new Error('能力不存在'), { status: 404 });
  if (!caps[idx].draft) throw Object.assign(new Error('没有草稿可弃'), { status: 400 });
  delete caps[idx].draft;
  caps[idx].updatedAt = new Date().toISOString();
  saveCapabilities(caps);

  appendAuditLog({
    action: 'discard_draft',
    capId, editedBy,
    detail: '弃稿',
  });
  return caps[idx];
}

// 发布：草稿→生效版
function publishDraft(capId, publishedBy) {
  const caps = listCapabilities();
  const idx = caps.findIndex(c => c.id === capId);
  if (idx === -1) throw Object.assign(new Error('能力不存在'), { status: 404 });
  if (!caps[idx].draft) throw Object.assign(new Error('没有草稿可发布'), { status: 400 });

  const cap = caps[idx];
  const oldVersion = cap.published ? cap.published.version : 0;
  const now = new Date().toISOString();

  // 旧版→历史
  const history = cap.history || [];
  if (cap.published) {
    history.push({ ...cap.published });
  }
  // 如果历史超过上限，截断最早的
  while (history.length > cap.maxHistory) {
    history.shift();
  }

  // 草稿→生效版
  cap.published = {
    version: cap.draft.version,
    content: JSON.parse(JSON.stringify(cap.draft.content)),
    publishedAt: now,
    publishedBy,
  };
  cap.draft = null;
  cap.history = history;
  cap.updatedAt = now;
  caps[idx] = cap;
  saveCapabilities(caps);

  appendAuditLog({
    action: 'publish',
    capId, publishedBy,
    version: cap.published.version,
    oldVersion,
    detail: '发布生效',
  });

  return cap;
}

// 回滚到历史版本
function rollbackToVersion(capId, targetVersion, publishedBy) {
  const caps = listCapabilities();
  const idx = caps.findIndex(c => c.id === capId);
  if (idx === -1) throw Object.assign(new Error('能力不存在'), { status: 404 });
  const cap = caps[idx];

  // 查找目标版本
  let targetContent = null;
  let targetVer = null;
  if (cap.published && cap.published.version === targetVersion) {
    targetContent = cap.published.content;
    targetVer = cap.published.version;
  }
  if (!targetContent && cap.history) {
    const found = cap.history.find(h => h.version === targetVersion);
    if (found) {
      targetContent = found.content;
      targetVer = found.version;
    }
  }
  if (!targetContent) throw Object.assign(new Error('目标版本不存在'), { status: 404 });

  // 回滚 = 用目标版本内容创建新版本草稿（不直接生效）
  const newVersion = cap.published ? cap.published.version + 1 : 1;
  cap.draft = {
    version: newVersion,
    content: JSON.parse(JSON.stringify(targetContent)),
    updatedAt: new Date().toISOString(),
    rollbackFrom: targetVer,
  };
  cap.updatedAt = new Date().toISOString();
  caps[idx] = cap;
  saveCapabilities(caps);

  appendAuditLog({
    action: 'rollback',
    capId, publishedBy,
    rollbackToVersion: targetVer,
    draftVersion: newVersion,
    detail: `回滚到 v${targetVer}`,
  });

  return cap;
}

// 获取审计日志
function getAuditLog(options = {}) {
  const logs = store.read(AUDIT_FILE, []);
  const { capId, limit = 50 } = options;
  let filtered = logs;
  if (capId) filtered = filtered.filter(l => l.capId === capId);
  return filtered.slice(-limit).reverse();
}

// 获取版本历史
function getVersionHistory(capId) {
  const cap = getCapability(capId);
  if (!cap) throw Object.assign(new Error('能力不存在'), { status: 404 });
  const versions = [];
  if (cap.published) {
    versions.push({
      version: cap.published.version,
      status: 'published',
      content: cap.published.content,
      publishedAt: cap.published.publishedAt,
      publishedBy: cap.published.publishedBy,
    });
  }
  if (cap.draft) {
    versions.push({
      version: cap.draft.version,
      status: 'draft',
      content: cap.draft.content,
      updatedAt: cap.draft.updatedAt,
    });
  }
  if (cap.history) {
    for (const h of cap.history) {
      versions.push({
        version: h.version,
        status: 'history',
        content: h.content,
        publishedAt: h.publishedAt,
        publishedBy: h.publishedBy,
      });
    }
  }
  return versions.sort((a, b) => (b.version || 0) - (a.version || 0));
}
```

```javascript
// 返回草稿版结果和当前生效版结果，供对比
function trialRun(capId, testQuestion, role, bizLine) {
  const cap = getCapability(capId);
  if (!cap) throw Object.assign(new Error('能力不存在'), { status: 404 });
  if (!cap.draft) throw Object.assign(new Error('没有草稿可试跑'), { status: 400 });

  // 获取 RAG 知识
  const ragEngine = require('./rag-engine');
  const ragResults = ragEngine.search(testQuestion, { role, bizLine, limit: 5 });

  // 用草稿版组装 prompt
  const draftPrompt = assemblePromptFromCap(cap.draft.content, testQuestion, role, bizLine, ragResults);
  const draftResult = generateMockResult(draftPrompt, role, testQuestion);

  // 用生效版组装 prompt
  const publishedPrompt = cap.published
    ? assemblePromptFromCap(cap.published.content, testQuestion, role, bizLine, ragResults)
    : null;
  const publishedResult = publishedPrompt
    ? generateMockResult(publishedPrompt, role, testQuestion)
    : null;

  return {
    draft: { prompt: draftPrompt, result: draftResult },
    published: publishedPrompt ? { prompt: publishedPrompt, result: publishedResult } : null,
    ragChunks: ragResults.map(r => ({ heading: r.heading, content: r.content, score: r.score })),
  };
}

// 用能力定义组装 prompt（与 prompt-engine 的 assemblePrompt 类似）
function assemblePromptFromCap(content, question, role, bizLine, ragChunks) {
  const sections = [
    `[角色与业务线] 你是一名${content.title || role}，服务于${bizLine}业务线。`,
    ``,
    `[角色设定] ${content.description || ''}`,
    ``,
    `[输出格式要求]`,
    content.outputFormat || '（无格式要求）',
    ``,
    `[RAG 召回的知识]`,
    (ragChunks && ragChunks.length > 0)
      ? ragChunks.map((c, i) => `[${i+1}] ${c.heading || ''}\n${c.content || ''}`).join('\n\n')
      : '（无 RAG 召回）',
    ``,
    `[用户问题]`,
    question || '',
  ];
  return sections.join('\n');
}
```

```javascript
// diff 辅助函数：简单文本差异标记
function diffTexts(a, b) {
  if (!a) return { added: b || '', removed: '' };
  if (!b) return { added: '', removed: a || '' };
  const aLines = a.split('\n');
  const bLines = b.split('\n');

  // 简单行级 diff（只标记有变化的行，不逐词对比）
  const added = [];
  const removed = [];
  const common = [];

  let i = 0, j = 0;
  while (i < aLines.length || j < bLines.length) {
    if (i < aLines.length && j < bLines.length && aLines[i] === bLines[j]) {
      common.push(aLines[i]);
      i++; j++;
    } else {
      if (i < aLines.length) removed.push(aLines[i++]);
      if (j < bLines.length) added.push(bLines[j++]);
    }
  }

  return { added: added.join('\n'), removed: removed.join('\n'), common: common.join('\n') };
}
```

```javascript
// routes/capability.js
const express = require('express');
const router = express.Router();
const capEngine = require('../lib/capability-engine');

// 中间件：检查登录
function requireAuth(req, res, next) {
  const user = req.headers['x-user'] ? JSON.parse(req.headers['x-user']) : null;
  if (!user) return res.status(401).json({ ok: false, error: '未登录' });
  req.user = user;
  next();
}

// 能力列表
router.get('/', requireAuth, (req, res) => {
  try {
    const summaries = capEngine.listCapabilitySummaries();
    res.json({ ok: true, capabilities: summaries });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
});

// 能力详情（含生效版和草稿版内容）
router.get('/:id', requireAuth, (req, res) => {
  try {
    const cap = capEngine.getCapability(req.params.id);
    if (!cap) return res.status(404).json({ ok: false, error: '能力不存在' });
    res.json({ ok: true, capability: cap });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
});

// 保存草稿
router.post('/:id/draft', requireAuth, (req, res) => {
  try {
    const cap = capEngine.editDraft(req.params.id, req.body.content, req.user.username || req.user.role || 'unknown');
    res.json({ ok: true, capability: cap });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
});

// 获取草稿内容
router.get('/:id/draft', requireAuth, (req, res) => {
  try {
    const draft = capEngine.getDraft(req.params.id);
    if (!draft) return res.json({ ok: true, draft: null });
    res.json({ ok: true, draft });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
});

// 弃稿
router.delete('/:id/draft', requireAuth, (req, res) => {
  try {
    const cap = capEngine.discardDraft(req.params.id, req.user.username || req.user.role || 'unknown');
    res.json({ ok: true, capability: cap });

});

  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }

// 回滚

  try {
    const { version } = req.body;
    if (!version) return res.status(400).json({ ok: false, error: '请指定回滚版本号' });
    const cap = capEngine.rollbackToVersion(req.params.id, Number(version), req.user.username || req.user.role || 'unknown');
    res.json({ ok: true, capability: cap });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
});

// 版本历史
router.get('/:id/history', requireAuth, (req, res) => {
  try {
    const versions = capEngine.getVersionHistory(req.params.id);
    res.json({ ok: true, versions });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
});

// 审计日志
router.get('/:id/audit', requireAuth, (req, res) => {
  try {
    const logs = capEngine.getAuditLog({ capId: req.params.id });
    res.json({ ok: true, auditLog: logs });
  } catch (e) {
    res.status(e.status || 500).json({ ok: false, error: e.message });
  }
```
